Minebot library

Simple Python automation library.

install

- install with pip and git
- pip install git+https://github.com/Andreaswinter2012/minebot.git and install Git for windows (winget install Git.Git -h)

Example 1

import minebot
import time

time.sleep(5)

start = True

while True:
    while start:
        minebot.removeblock(3)

        if minebot.chat("r", "stop.png"):
            start = False
            break

    if minebot.chat("r", "start.png"):
        start = True

Example 2

import minebot
import time

count = 0
countin = int(input("count: "))

time.sleep(5)

while count < countin:
    count += 1
    minebot.removeblock(1)
    time.sleep(1.5)

minebot.walk("d", 1)
minebot.select("2")
minebot.drop("all")
minebot.walk("a", 1)
minebot.select("1")

Example 3

import time
import minebot
countin = int(input("count:"))
count = 0
time.sleep(5)
while count < countin:
    count += 1
    minebot.removeblock(1.2)
    minebot.walk("w", 0.2)
minebot.chat("w", "/tp Андрей")
time.sleep(2)
for i in range(9):
    minebot.select(str(i+1))
    minebot.drop("all")
minebot.select("1")

All functions

removeblock(time)       # removeblock(2)
walk(key, time)         # walk("w", 1)
select(slot)            # select(9) 
drop(mode)              # drop() or drop("all")
chat(mode, image)       # chat("w", "Hello World") or chat("r", "image.png")
press(key)              # press("enter")
screen(file, conf)      # screen("image.png", 0.9) and return True or False
click(mod, x, y)        # click("l", 500, 500) or click("r", 500, 500) or click("l")
find(file, conf)        # find("image.png", 0.9) and return x, y simple (148 167)
jump(sleep)             # jump(2)
setblock()              # setblock()
